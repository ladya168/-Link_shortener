    image: postgres:18
    environment:
      POSTGRES_DB: postgres
      POSTGRES_USER: postgres

    hostname: db
    ports:
      - "6432:5432"
    networks:
      - project

      interval: 2s
      timeout: 2s
      retries: 10
    volumes:
      - postgres_data:/var/lib/postgresql
  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
    networks:
      - project
    volumes:
      - './nginx.conf:/etc/nginx/nginx.conf'
    depends_on:
      - project_back
      - project_front

  project_back:
    build: .
    networks:
      - project
    depends_on:
      db:
        condition: service_healthy
